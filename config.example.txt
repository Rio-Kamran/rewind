# Rewind settings. Change a line, then right-click the tray icon -> Reload config.
# Lines starting with # are comments.

# Press this to save the last <seconds> as a clip. Examples: ctrl+alt+p, F9, shift+F10
hotkey=ctrl+alt+p

# A second key that saves just the last <short_seconds>. off = no second key.
hotkey_short=ctrl+alt+o

# Starts a long recording; press again to stop and save it. off = no key.
hotkey_record=ctrl+alt+r

# Saves the latest frame as a PNG (and copies it, ready to paste). off = no key.
hotkey_screenshot=ctrl+alt+i

# How far back a clip reaches, in seconds (5-600). Memory use is about bitrate x seconds / 8 MB.
seconds=60

# How far back the short clip reaches, in seconds (3 up to seconds-1).
short_seconds=15

# Frames per second to record (15-240).
fps=60

# Video quality in megabits per second (2-150). 20 is plenty for 1440p60 h264.
bitrate_mbps=20

# h264 plays everywhere. av1 = smaller files, same quality, needs a newer phone to play.
codec=h264

# primary = the main monitor. Or a number: run  Rewind.exe --list  to see them.
monitor=primary

# Record what comes out of the speakers/headset (track 1).
game_audio=on

# Record the microphone as its own track (track 2).
mic=on

# Cleanup applied to the mic (an ffmpeg audio filter). off = raw mic.
mic_filter=afftdn=nr=12:nf=-40

# If sound lands late or early against the picture, shift it here (milliseconds, + = later).
audio_offset_ms=0

# always = record all the time. games = only while a game is in front (see games= below).
record=always

# Apps that count as a game even in a window (process names, comma separated). Any app covering the whole monitor counts too.
games=javaw, FortniteClient-Win64-Shipping, RobloxPlayerBeta, RocketLeague, GeometryDash, Minecraft.Windows

# In games mode: how long a game can be out of front before recording pauses (5-600 s).
game_grace_seconds=45

# A long recording stops itself after this many minutes (1-600). About 150 MB per minute at 20 Mbps.
recording_max_minutes=120

# Copy for Discord shrinks a clip to fit under this many MB (1-500). Discord's free limit is 20.
share_max_mb=20

# Delete the oldest clips once the clips folder passes this many GB. 0 = never delete anything.
max_storage_gb=0

# on = a small "Clip saved" card on screen (shows over most games). off = tray balloons only.
toast=on

# Play a chime when a clip, screenshot or recording lands. Drop your own clip.wav next to Rewind.exe to change it.
sound=on

# How loud the chime is (0-100).
sound_volume=80

# on = saying the phrase below into the mic saves a clip, like Medal's "clip that".
voice_clip=on

# What to say. Two or three clear words work best.
voice_phrase=clip that

# windows = the recogniser built into Windows (offline). riovoice = a RioVoice speech-to-text server (more accurate).
voice_engine=windows

# The RioVoice streaming address (only used with voice_engine=riovoice).
voice_url=wss://thoughts.riomax.com/ws/transcribe

# Where clips go. Leave empty for your own Videos\Rewind folder.
clips=

# Leave empty to use the ffmpeg on PATH, or give a full path to ffmpeg.exe.
ffmpeg=

# on = every 6 hours, fetch a newer Rewind from GitHub, check its SHA-256 and restart into it once nothing is recording. off = never.
auto_update=on
